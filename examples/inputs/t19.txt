def compute(a, b, c, d, e) {
t1 = a * b + c * d ;
t2 = t1 - b * c + a * d ;
t3 = t2 * e - t1 + t2 * a ;
t4 = t3 + compute(t1, t2, t3, a, b) ;
return t4 * e + t3 - t2 * t1 ;
}
